std::vector<std::vector<std::string>> CSVLoader::split_data(std::string raw)
{
    std::vector<std::vector<std::string>> items;

    std::string item;
    size_t row;
    size_t col;
    for(char character : raw)
    {
        switch (character)
        {
            case ',':
            {
                items[row].push_back(item);
                item.clear();

                col++;

                break;
            }

            case '\n':
            {
                items[row].push_back(item);
                item.clear();

                col = 0;
                row++;
            }

            default:
            {
                // item.append(character);
                item += character;
                break;
            }
        }
    }
    
}
